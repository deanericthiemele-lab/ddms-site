$tomlPath = "netlify.toml"
if (Test-Path $tomlPath) {
    $content = Get-Content $tomlPath -Raw
    $cleaned = $content -replace '(?ms)\[\[edge_functions\]\]\s*\r?\npath\s*=\s*"/\*"\s*\r?\nfunction\s*=\s*"basic-auth"\s*\r?\n?', ''
    Set-Content -Path $tomlPath -Value $cleaned -NoNewline
    Write-Host "OK : netlify.toml nettoye"
} else {
    Write-Host "netlify.toml introuvable ici"
}

$edgeFuncDir = "netlify\edge-functions"
if (Test-Path $edgeFuncDir) {
    Get-ChildItem -Path $edgeFuncDir -Filter "basic-auth.*" | ForEach-Object {
        Remove-Item $_.FullName -Force
        Write-Host "OK : supprime $($_.Name)"
    }
} else {
    Write-Host "Pas de dossier netlify\edge-functions"
}
